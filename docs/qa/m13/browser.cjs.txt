const {chromium}=require('/usr/local/lib/python3.13/site-packages/playwright/driver/package');const fs=require('fs');
const P='/data/audit13',base='http://127.0.0.1:18173';
(async()=>{let browser;let cid;try{
const ids=JSON.parse(fs.readFileSync(P+'/browser-ready.json')).cameras;cid=ids[0];
let r=await fetch('http://127.0.0.1:18100/api/v1/cameras/'+cid+'/start',{method:'POST'});if(!r.ok)throw Error('Start failed');
await new Promise(r=>setTimeout(r,12000));
browser=await chromium.launch({headless:true,executablePath:'/usr/local/bin/chromium',args:['--no-sandbox']});const ctx=await browser.newContext({acceptDownloads:true});const page=await ctx.newPage();let errors=[],frames=0;page.on('pageerror',e=>errors.push(e.message));page.on('websocket',ws=>ws.on('framereceived',()=>frames++));
const events=await (await fetch('http://127.0.0.1:18100/api/v1/events?limit=10')).json();if(!events.items.length)throw Error('No alarms');
const routes=['/monitoring','/cameras','/events','/analytics','/settings','/monitoring/'+cid,'/events/'+events.items[0].id];let checks=[];
for(const width of [1280,390]){await page.setViewportSize({width,height:900});for(const theme of ['light','dark']){await page.goto(base+'/monitoring',{waitUntil:'domcontentloaded'});await page.evaluate(t=>{localStorage.setItem('ai-detector.theme',t)},theme);for(const route of routes){await page.goto(base+route,{waitUntil:'domcontentloaded'});await page.waitForTimeout(route.includes(cid)?1200:550);const d=await page.evaluate(()=>({overflow:Math.max(document.documentElement.scrollWidth,document.body.scrollWidth)-innerWidth,text:document.querySelector('main')?.innerText.length??document.body.innerText.length}));if(d.overflow>1||d.text<20)throw Error('Layout '+route+' '+JSON.stringify(d));checks.push({route,width,theme,...d});if(width===390&&theme==='dark'&&route==='/monitoring/'+cid)await page.screenshot({path:P+'/mobile-focus-dark.png',fullPage:true});if(width===1280&&theme==='light'&&route==='/analytics')await page.screenshot({path:P+'/desktop-analytics.png',fullPage:true});}}}
if(frames<5)throw Error('No WS preview packets');
await page.goto(base+'/events');await page.getByRole('button',{name:'Скачать CSV',exact:true}).waitFor();let [download]=await Promise.all([page.waitForEvent('download'),page.getByRole('button',{name:'Скачать CSV',exact:true}).click()]);await download.saveAs(P+'/browser-events.csv');
await page.route('**/api/v1/events/export*',route=>route.fulfill({status:503,contentType:'application/json',body:JSON.stringify({error:{code:'SERVICE_UNAVAILABLE',message:'QA outage'}})}));await page.getByRole('button',{name:'Скачать CSV',exact:true}).click();await page.getByRole('alert').waitFor();await page.unroute('**/api/v1/events/export*');
await page.route('**/api/v1/analytics?*',route=>route.fulfill({status:503,contentType:'application/json',body:JSON.stringify({error:{code:'SERVICE_UNAVAILABLE',message:'QA outage'}})}));await page.goto(base+'/analytics');await page.getByRole('alert').first().waitFor();await page.unroute('**/api/v1/analytics?*');await page.getByRole('button',{name:'Повторить',exact:true}).first().click();await page.waitForTimeout(800);
const blocked=await browser.newContext();await blocked.addInitScript(()=>{Storage.prototype.getItem=()=>{throw new DOMException('blocked','SecurityError')};Storage.prototype.setItem=()=>{throw new DOMException('blocked','SecurityError')}});const bp=await blocked.newPage();bp.on('pageerror',e=>errors.push(e.message));await bp.goto(base+'/monitoring');await bp.waitForTimeout(700);if((await bp.locator('main').innerText()).length<20)throw Error('Blocked storage blank UI');await blocked.close();
if(errors.length)throw Error(JSON.stringify(errors));fs.writeFileSync(P+'/browser-proof.json',JSON.stringify({route_layout_checks:checks,pageerrors:errors,ws_frames:frames,csv_download:true,export_503_error:true,analytics_503_retry:true,blocked_storage_render:true},null,2));console.log(JSON.stringify({checks:checks.length,pageerrors:errors.length,ws_frames:frames,csv_download:true,error_retry:true}));
await ctx.close();await browser.close();browser=null;
r=await fetch('http://127.0.0.1:18100/api/v1/cameras/'+cid+'/stop',{method:'POST'});if(!r.ok)throw Error('Stop failed');await new Promise(r=>setTimeout(r,1000));fs.writeFileSync(P+'/browser-done','ok');process.exit(0);
}catch(e){console.error(e);if(browser)await browser.close();process.exit(1)}})();
