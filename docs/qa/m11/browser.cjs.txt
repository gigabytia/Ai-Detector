const {chromium}=require('playwright');const assert=require('node:assert/strict');const fs=require('node:fs');
const out='/data/m10/Ai-Detector/docs/qa/m11';const target=JSON.parse(fs.readFileSync(out+'/targets.json'));
(async()=>{const browser=await chromium.launch({executablePath:'/usr/local/bin/chromium',headless:true,args:['--no-sandbox']});try{
 const page=await browser.newPage({viewport:{width:1440,height:1000},acceptDownloads:true});const errors=[];page.on('pageerror',e=>errors.push(e.message));
 await page.goto('http://127.0.0.1:15173/events');await page.getByRole('button',{name:'Скачать CSV',exact:true}).waitFor();
 let selects=page.locator('select');await selects.nth(0).selectOption(target.camera);await selects.nth(1).selectOption('FALSE_POSITIVE');await selects.nth(2).selectOption('PERSON_LOITERING_IN_DANGER_ZONE');await selects.nth(3).selectOption('24');
 await page.locator(`[data-event-id="${target.event}"]`).waitFor();const [download]=await Promise.all([page.waitForEvent('download'),page.getByRole('button',{name:'Скачать CSV',exact:true}).click()]);await download.saveAs('/data/m11/browser-export.csv');assert.equal(download.suggestedFilename(),'events.csv');
 await page.screenshot({path:out+'/events-desktop.png',fullPage:true});await page.setViewportSize({width:390,height:844});assert(await page.evaluate(()=>document.documentElement.scrollWidth<=innerWidth));await page.screenshot({path:out+'/events-mobile-light.png',fullPage:true});await page.evaluate(()=>document.documentElement.classList.add('dark'));await page.screenshot({path:out+'/events-mobile-dark.png',fullPage:true});
 await page.route('**/api/v1/events/export?**',route=>route.fulfill({status:503,contentType:'application/json',body:'{"error":{"code":"SERVICE_UNAVAILABLE","message":"QA"}}'}));await page.getByRole('button',{name:'Скачать CSV',exact:true}).click();await page.getByRole('alert').filter({hasText:'Файл не создан'}).waitFor();await page.screenshot({path:out+'/export-error.png',fullPage:true});await page.unroute('**/api/v1/events/export?**');const [retry]=await Promise.all([page.waitForEvent('download'),page.getByRole('button',{name:'Скачать CSV',exact:true}).click()]);assert.equal(retry.suggestedFilename(),'events.csv');
 await page.goto(`http://127.0.0.1:15173/events/${target.event}`);await page.getByText('Непрерывное наблюдение:',{exact:false}).waitFor();assert(await page.evaluate(()=>document.documentElement.scrollWidth<=innerWidth));await page.screenshot({path:out+'/loitering-detail-mobile.png',fullPage:true});
 await page.goto(`http://127.0.0.1:15173/monitoring/${target.camera}`);const field=page.locator(`#loitering_threshold_seconds-${target.camera}`);await field.waitFor();await field.fill('2');await page.getByRole('button',{name:'Сохранить',exact:true}).click();await page.getByText('Настройки сохранены',{exact:true}).waitFor();const config=await (await page.request.get(`http://127.0.0.1:18000/api/v1/cameras/${target.camera}/config`)).json();assert.equal(config.effective.loitering_threshold_seconds,2);
 assert.deepEqual(errors,[]);console.log(JSON.stringify({csvDownload:true,matchingCameraStatusTypePeriod:true,retryAfter503:true,loiteringDetail:true,thresholdSavedViaUi:2,mobileOverflow:false,pageerrors:errors}));
}finally{await browser.close()}process.exit(0)})().catch(e=>{console.error(e);process.exit(1)});
