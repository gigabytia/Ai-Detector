import asyncio,json,os,signal,time,subprocess
from pathlib import Path
from uuid import UUID
import httpx,websockets
from redis.asyncio import Redis
from sqlalchemy import text
from sqlalchemy.ext.asyncio import create_async_engine

def pid(name):
 lines=subprocess.check_output(['ps','-eo','pid,args'],text=True).splitlines()
 return next(int(line.split()[0]) for line in lines if '/.venv/bin/'+name in line and 'uv run' not in line)
async def main():
 worker,api=pid('ai-detector-worker'),pid('ai-detector-api')
 engine=create_async_engine('postgresql+asyncpg://ai_detector@127.0.0.1:15432/ai_detector_m10')
 redis=Redis.from_url('redis://127.0.0.1:16379/0')
 async with httpx.AsyncClient(base_url='http://127.0.0.1:18000',timeout=10) as http:
  cams=(await http.get('/api/v1/cameras')).json();camera=next(c['id'] for c in cams if c['enabled'])
  statuses=[json.loads(await redis.get(k)) for k in await redis.keys('ad:camera:*:status')]
  runs=[UUID(s['run_id']) for s in statuses];assert len(runs)==3
  async with engine.connect() as conn:
   before=set(str(x) for x in (await conn.execute(text('SELECT id FROM track_sessions'))).scalars())
  ready=asyncio.Event()
  async def sse():
   try:
    async with http.stream('GET','/api/v1/events/stream') as response:
     async for line in response.aiter_lines():
      if line:ready.set()
   except httpx.HTTPError:pass
  stream=asyncio.create_task(sse());await asyncio.wait_for(ready.wait(),3)
  async with websockets.connect(f'ws://127.0.0.1:18001/cameras/{camera}/telemetry') as ws:
   await asyncio.wait_for(ws.recv(),3);started=time.monotonic();os.kill(worker,signal.SIGTERM)
   try:
    while True:await asyncio.wait_for(ws.recv(),5)
   except websockets.ConnectionClosed:pass
  for _ in range(120):
   if not Path(f'/proc/{worker}').exists():break
   await asyncio.sleep(.05)
  worker_time=round(time.monotonic()-started,2)
  assert not Path(f'/proc/{worker}').exists()
  assert not await redis.keys('ad:camera:*:status') and not await redis.keys('ad:worker:*:heartbeat')
  for _ in range(100):
   if await redis.xlen('ad:ingest')==await redis.xlen('ad:analytics:ingest')==0:break
   await asyncio.sleep(.05)
  assert await redis.xlen('ad:ingest')==await redis.xlen('ad:analytics:ingest')==0
  async with engine.connect() as conn:
   saved=list((await conn.execute(text('SELECT id,run_id FROM track_sessions WHERE run_id=ANY(:runs)'),{'runs':runs})).all())
  new=[x for x in saved if str(x.id) not in before];assert new
  # At least one still-active track is finalized on stop; all sessions from current runs persist.
  dashboard=(await http.get('/api/v1/analytics')).json();assert dashboard['active_cameras']==0
  events=(await http.get('/api/v1/events?limit=50')).json()['items'];snapshots=[e for e in events if e.get('snapshot_url')];assert snapshots
  for event in snapshots[:5]:assert (await http.get(event['snapshot_url'])).status_code==200
  started=time.monotonic();os.kill(api,signal.SIGTERM);await asyncio.wait_for(stream,6)
  for _ in range(100):
   if not Path(f'/proc/{api}').exists():break
   await asyncio.sleep(.05)
  assert not Path(f'/proc/{api}').exists()
  print(json.dumps({'worker_shutdown_seconds':worker_time,'api_with_sse_shutdown_seconds':round(time.monotonic()-started,2),'runtime_keys_removed':True,'event_stream_backlog':0,'analytics_stream_backlog':0,'new_completed_sessions_on_stop':len(new),'current_runs_sessions_saved':len(saved),'recent_snapshot_reads_after_worker_stop':min(len(snapshots),5)}),flush=True)
 await engine.dispose();await redis.aclose()
asyncio.run(main())
